cmake_minimum_required(VERSION 3.20)
project(zeta_asr_qualification LANGUAGES C CXX)
if(NOT DEFINED WHISPER_SOURCE)
  message(FATAL_ERROR "Pass the verified, pinned WHISPER_SOURCE directory")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper)
add_executable(asr-benchmark benchmark.cpp)
target_compile_features(asr-benchmark PRIVATE cxx_std_17)
target_compile_options(asr-benchmark PRIVATE -Wall -Wextra -Werror)
target_link_libraries(asr-benchmark PRIVATE whisper)

add_executable(zeta-whisper-helper helper.cpp)
target_compile_features(zeta-whisper-helper PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(zeta-whisper-helper PRIVATE /W4 /WX)
else()
  target_compile_options(zeta-whisper-helper PRIVATE -Wall -Wextra -Werror)
endif()
target_link_libraries(zeta-whisper-helper PRIVATE whisper)
